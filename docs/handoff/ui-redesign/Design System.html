Emboss

Three variants. Stacked inset + drop shadows create the carved crown-molded look.

/* Neutral — stepper, bulk chips, cancel */
--of-btn-emboss:
  inset 0 1px 0 rgba(255, 245, 220, 0.65),     /* top highlight */
  inset 0 2px 4px rgba(74, 46, 21, 0.18),      /* inset top shadow */
  inset 0 -2px 0 rgba(0, 0, 0, 0.14),          /* bottom shadow */
  inset 0 0 0 1px rgba(94, 60, 24, 0.28),      /* inner rim */
  0 1px 0 rgba(255, 245, 220, 0.45),           /* outer highlight */
  0 2px 3px rgba(74, 46, 21, 0.20),            /* outer shadow */
  0 4px 6px rgba(74, 46, 21, 0.12);            /* soft drop */

/* Strong — rust primary CTAs */
--of-btn-emboss-strong: /* same stack, brighter highlights + heavier drop */

/* Active — pressed state */
--of-btn-emboss-active:
  inset 0 3px 5px rgba(0, 0, 0, 0.28),         /* deep recess */
  inset 0 -1px 0 rgba(255, 245, 220, 0.20),    /* bottom catch-light */
  inset 0 0 0 1px rgba(94, 60, 24, 0.40);

/* Progress channel — recessed for sitting-bar look */
--of-channel-emboss: /* same flavor as -active but with outer highlight */
